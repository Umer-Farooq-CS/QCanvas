set AWS_REGION=us-east-1
for /f "delims=" %A in ('aws sts get-caller-identity --query Account --output text') do set AWS_ACCOUNT_ID=%A

aws ecr get-login-password --region %AWS_REGION% | docker login --username AWS --password-stdin %AWS_ACCOUNT_ID%.dkr.ecr.%AWS_REGION%.amazonaws.com
docker build -t qcanvas-api:latest -f backend/Dockerfile .
docker tag qcanvas-api:latest %AWS_ACCOUNT_ID%.dkr.ecr.%AWS_REGION%.amazonaws.com/qcanvas-api:latest
docker push %AWS_ACCOUNT_ID%.dkr.ecr.%AWS_REGION%.amazonaws.com/qcanvas-api:latest
